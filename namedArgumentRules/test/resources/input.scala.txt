package example

final case class Record(name: String, count: Int = 1)
final class Contextual(val value: Int)(using val suffix: String)
final class Curried(val first: Int)(val second: Int)
object Other:
  def header(pkg: String, enabled: Boolean = true): String = if enabled then pkg else ""
  def overloaded(value: String): String = value
  def overloaded(number: Int): Int = number
  def curried(first: Int)(second: Int): Int = first + second
  def generic[A](value: A): A = value
  def choose(flag: Boolean)(value: => Int): Int = if flag then value else 0
  def contextual(value: Int)(using suffix: String): String = s"$value$suffix"
  def pair(first: String, second: String): String = first + second
  def repeated(values: Int*): Int = values.sum
  def keyword(`type`: String): String = `type`
  def returned(value: Int): Int => Int = other => value + other
  extension (value: String)
    def decorated(prefix: String): String = prefix + value
